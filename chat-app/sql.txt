create table coaches (
  id serial primary key,
  name varchar(100),
  password varchar(100),
  age int,
  exp_years int
);

create table clients (
  id serial primary key,
  name varchar(100),
  password varchar(100),
  age int,
  coach_id integer references coaches(id)
);

create table users (
  id serial primary key,
  name varchar(100),
  password varchar(100),
  role varchar(50),
  age int
);
