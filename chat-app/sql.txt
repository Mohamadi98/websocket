create table users (
  id serial primary key,
  name varchar(100),
  password varchar(100),
  role varchar(50),
  age int
);

create table messages (
  id serial primary key,
  sender_id int,
  recipient_id int,
  message varchar(255),
  created_at timestamptz default NOW()
);

create table last_seen (
  id serial primary key,
  main_participant_id int,
  participant_id int,
  seen_at timestamptz
);
